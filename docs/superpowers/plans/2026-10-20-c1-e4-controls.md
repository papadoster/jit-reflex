Spec: controls spec §3.1 and §13 items 1–5. Paper formulas (arXiv 2605.11459 §3.4, App. A.10):
- τ_N = 1[max(0, cos(v_N, v_{N−1})) < 1/2], with τ_N = 0 when either reading has norm ≤ `ppc_v_min` (our convention);
- C_N = β_out τ_N + (1−β_out) C_{N−1};
- s_N = C_N/(C_N+R_TH), on the updated C;
- L_N = β_in + (1−β_in) L_{N−1} if τ_N else [1 − β_in(1−s_N)] L_{N−1};
- m_N = L_N > L_th;
- β_in 0.3, β_out = 1 − 2^(−2/16), L_th = R_TH = 0.3·0.7²; while m_N the next call ≤ t_obs,N + 4;
- reset (L, C) ← 0 when ‖EEF − p‖ < 0.03 m.

The reading v_N = p(t_obs,N) − p(t_obs,N − 1).

def _latch_agent(readings, th=None):
    """A PPCL agent whose chunk resets read the given object velocities (one per reset, m/step), 10 steps apart, the
    object far from the arm; returns the agent and the latched flag after each reset."""
    ag = orx.Agent("PPCL", 10, 0)
    if th is not None:
        ag.latch_th = th
    p, on = np.array([0.5, 0.0, 0.0]), []
    for n, v in enumerate(readings):
        t = 10 * n + 1
        ag.p_hist[t - 1], ag.p_hist[t] = p.copy(), p + np.asarray(v, float)
        p = p + np.asarray(v, float)
        ag.sched.t_obs = t
        ag.on_new_chunk()
        on.append(ag.latch_next < math.inf)
    return ag, on

def _ppc_run(method, path, th=None, s=10, steps=60):
    """PPC / PPCL on one object path (positions by step), the arm walking to the seen object, d = 0."""
    ag = orx.Agent(method, s, 0)
    if th is not None:
        ag.latch_th = th
    x, acts, calls = np.zeros(3), [], []
    for t in range(steps):
        ag.observe(t, x, path[t])
        if ag.sched.arrive(t):
            ag.on_new_chunk()
        if ag.sched.wants_call(t, ag.trigger(t)):
            calls.append(t)
            ag.sched.issue(t, _brain(x, path[t]))
            if ag.sched.arrive(t):
                ag.on_new_chunk()
        a = ag.act(t, ag.sched.action(t))
        acts.append(a)
        x = x + orx.G_POS * a[:3]
    return np.array(acts), calls, ag


def _wiggle(seed, steps=60):
    rng = np.random.default_rng(seed)
    v = rng.normal(0, 0.003, (steps, 3)) * np.array([1, 1, 0])  # irregular motion in the table plane
    return np.array([0.30, 0.0, 0.0]) + np.cumsum(v, axis=0)


def test_ppcl_without_latch_is_ppc_bitwise():

```python
    def _latch_update(self):
        """PPCL at a chunk reset (controls spec §3.1, the paper's App. A.10): the reading v_N = p(t_obs) - p(t_obs - 1)
        of the plan now starting; tau = 1 when cos(v_N, v_N-1) < 1/2, both readings above ppc_v_min (a still reading
        never triggers: our convention, the paper's cosine is undefined there); the outer EMA C, the sticky factor on
        the updated C, the inner EMA L; latched (L > L_th): the next call at t_obs + 4 at the latest."""
        t = self.sched.t_obs
        v = self.p_hist[t] - self.p_hist[t - 1] if t - 1 in self.p_hist else None
        v = v if v is not None and np.linalg.norm(v) > self.ppc_v_min else None
        prev, self.latch_v = self.latch_v, v
        tau = (prev is not None and v is not None
               and max(0.0, float(v @ prev) / (np.linalg.norm(v) * np.linalg.norm(prev))) < 0.5)
        self.latch_C = PPCL_BETA_OUT * tau + (1 - PPCL_BETA_OUT) * self.latch_C
        s = self.latch_C / (self.latch_C + PPCL_R_TH)
        self.latch_L = (PPCL_BETA_IN + (1 - PPCL_BETA_IN) * self.latch_L if tau
                        else (1 - PPCL_BETA_IN * (1 - s)) * self.latch_L)
        on = self.latch_L > self.latch_th
        self.latch_next = t + PPCL_K_CAP if on else math.inf
        self.latch_stats["trig"] += tau
        self.latch_stats["chunks"] += on

    def _latch_reset(self):
        """PPCL's near-object reset: (L, C) <- 0 and the latch off; pace, path, offsets and the last reading stay."""
        self.latch_stats["resets"] += bool(self.latch_L or self.latch_C)
        self.latch_C = self.latch_L = 0.0
        self.latch_next = math.inf
```
